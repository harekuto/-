def method_bounds(src, signature):
    start = src.find(signature)
    if start < 0:
        raise SystemExit(f'method not found: {signature!r}')
    brace = src.find('{', start)
    depth = 0
    for i in range(brace, len(src)):
        if src[i] == '{':
            depth += 1
        elif src[i] == '}':
            depth -= 1
            if depth == 0:
                return start, i + 1
    raise SystemExit(f'unbalanced method: {signature!r}')

def replace_method(src, signature, replacement):
    start, end = method_bounds(src, signature)
    return src[:start] + replacement + src[end:]

def insert_after_method(src, anchor_signature, method_signature, method_text):
    if method_signature in src:
        return src
    _, end = method_bounds(src, anchor_signature)
    return src[:end] + '\n\n' + method_text + src[end:]
