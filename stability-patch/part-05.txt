+package net.execheinz.upgrader.economy;

+/** Pure, deterministic economy math kept free of Minecraft state for fuzz testing. */
+public final class EconomyMath {
+    private EconomyMath() {}
+
+    public static double upgradeChance(double inputValue, double targetValue,
+                                       double houseEdge, double minChance, double maxChance) {
+        if (!Double.isFinite(inputValue) || !Double.isFinite(targetValue)
+            || !Double.isFinite(houseEdge) || !Double.isFinite(minChance) || !Double.isFinite(maxChance)
+            || inputValue <= 0D || targetValue <= 0D || houseEdge <= 0D) {
+            return 0D;
+        }
+        double lo = Math.max(0D, Math.min(1D, Math.min(minChance, maxChance)));
+        double hi = Math.max(lo, Math.min(1D, Math.max(minChance, maxChance)));
+        double raw = houseEdge * inputValue / targetValue;
+        if (!Double.isFinite(raw) || raw <= 0D) return 0D;
+        return Math.max(lo, Math.min(hi, raw));
+    }
+
+    public static long sellPayout(double stackValue, double sellRate, long hardCap) {
+        if (!Double.isFinite(stackValue) || !Double.isFinite(sellRate)
+            || stackValue <= 0D || sellRate <= 0D || hardCap <= 0L) {
+            return 0L;
+        }
+        double rate = Math.max(0D, Math.min(1D, sellRate));
+        double raw = stackValue * rate;
+        if (!Double.isFinite(raw) || raw <= 0D) return 0L;
+        double bounded = Math.min(raw, Math.min((double) hardCap, (double) Long.MAX_VALUE / 4D));
+        return Math.max(0L, (long) Math.floor(bounded));
+    }
+
+    public static long saturatingAddPositive(long current, long amount, long cap) {
+        long safeCap = Math.max(0L, cap);
+        long safeCurrent = Math.max(0L, Math.min(current, safeCap));
+        if (amount <= 0L || safeCurrent >= safeCap) return safeCurrent;
+        if (amount >= safeCap - safeCurrent) return safeCap;
+        return safeCurrent + amount;
+    }
+
+    public static long remainingCapacity(long current, long cap) {
+        long safeCap = Math.max(0L, cap);
+        long safeCurrent = Math.max(0L, Math.min(current, safeCap));
+        return safeCap - safeCurrent;
+    }
+
+    public static int rewardCount(double targetValue, double unitValue, int maxStack) {
+        if (!Double.isFinite(targetValue) || !Double.isFinite(unitValue)
+            || targetValue <= 0D || unitValue <= 0D || maxStack <= 0) {
+            return 1;
+        }
+        double ratio = targetValue / unitValue;
+        if (!Double.isFinite(ratio) || ratio <= 1D) return 1;
+        long rounded = Math.round(Math.min(ratio, Integer.MAX_VALUE));
+        return (int) Math.max(1L, Math.min((long) maxStack, rounded));
+    }
